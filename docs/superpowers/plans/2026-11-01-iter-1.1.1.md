// Package logging 提供带大小滚动的日志写入器，供面板运行日志落盘使用。
package logging

import (
	"fmt"
	"os"
	"path/filepath"
)

// RotatingWriter 写入 <dir>/<name>：单文件超过 maxSize 时滚动为 .1 … .maxFiles，
// 超出份数的最旧文件被删除。写入不缓冲，进程退出无需 Close。
type RotatingWriter struct {
	dir      string
	name     string
	maxSize  int64
	maxFiles int
	f        *os.File
	size     int64
}

func NewRotatingWriter(dir, name string, maxSize int64, maxFiles int) *RotatingWriter {
	return &RotatingWriter{dir: dir, name: name, maxSize: maxSize, maxFiles: maxFiles}
}

func (w *RotatingWriter) path() string { return filepath.Join(w.dir, w.name) }

// Write 实现 io.Writer。
func (w *RotatingWriter) Write(p []byte) (int, error) {
	if err := w.ensureOpen(); err != nil {
		return 0, err
	}
	if w.size+int64(len(p)) > w.maxSize {
		if err := w.rotate(); err != nil {
			return 0, err
		}
		if err := w.ensureOpen(); err != nil {
			return 0, err
		}
	}
	n, err := w.f.Write(p)
	w.size += int64(n)
	return n, err
}

func (w *RotatingWriter) ensureOpen() error {
	if w.f != nil {
		return nil
	}
	if err := os.MkdirAll(w.dir, 0o755); err != nil {
		return fmt.Errorf("创建日志目录 %s 失败: %w", w.dir, err)
	}
	f, err := os.OpenFile(w.path(), os.O_CREATE|os.O_WRONLY|os.O_APPEND, 0o644)
	if err != nil {
		return fmt.Errorf("打开日志文件 %s 失败: %w", w.path(), err)
	}
	st, err := f.Stat()
	if err != nil {
		_ = f.Close()
		return err
	}
	w.f, w.size = f, st.Size()
	return nil
}

// rotate 关闭当前文件后，把 .(maxFiles-1)→.max、…、.1→.2、本体→.1 依次后移。
func (w *RotatingWriter) rotate() error {
	if w.f != nil {
		if err := w.f.Close(); err != nil {
			w.f = nil
			w.size = 0
			return err
		}
		w.f = nil
		w.size = 0
	}
	_ = os.Remove(fmt.Sprintf("%s.%d", w.path(), w.maxFiles))
	for i := w.maxFiles - 1; i >= 1; i-- {
		src := fmt.Sprintf("%s.%d", w.path(), i)
		if _, err := os.Stat(src); err != nil {
			continue
		}
		if err := os.Rename(src, fmt.Sprintf("%s.%d", w.path(), i+1)); err != nil {
			return fmt.Errorf("滚动日志失败: %w", err)
		}
	}
	if err := os.Rename(w.path(), w.path()+".1"); err != nil {
		return fmt.Errorf("滚动日志失败: %w", err)
	}
	return nil
}
```

package update

import (
	"fmt"
	"runtime"

// 升级引导：容器环境无法在页面里代用户执行命令，所以把"下一步敲什么"算成结构化数据，
// 前端只负责渲染与复制。版本号/包名/链接一律在此拼好，非法版本号直接不生成。
const (
	repoURL     = "https://github.com/busfly/frpc-mesh"
	dockerAsset = "frpc-mesh-v%s-docker-linux-arm64.tar.gz"
	// inspectMounts 在宿主机侧打印真实挂载，用户据此抄自己的数据目录（容器内看不到宿主机路径）
	inspectMounts = `docker inspect frpc-mesh --format '{{ range .Mounts }}-v {{ .Source }}:{{ .Destination }} {{ end }}'`
)

// GuideStep 是一步引导：标题 + 一句大白话说明 + 可复制命令 + 可选链接
type GuideStep struct {
	Title    string     `json:"title"`
	Desc     string     `json:"desc,omitempty"`
	Commands []string   `json:"commands,omitempty"`
	Link     *GuideLink `json:"link,omitempty"`
	Optional bool       `json:"optional,omitempty"`
}

// GuideLink 指向外部下载链接；URL 以 # 开头表示页内动作（如滚动到备份卡片）
type GuideLink struct {
	Label string `json:"label"`
	URL   string `json:"url"`
}

// UpgradeGuide 容器环境的完整升级引导
type UpgradeGuide struct {
	Kind     string      `json:"kind"` // docker_image | source_build
	Intro    string      `json:"intro"`
	Steps    []GuideStep `json:"steps"`
	Fallback GuideStep   `json:"fallback"`
	Note     string      `json:"note"`
}

func buildGuide(latestVer, arch string) *UpgradeGuide {
	if !versionRe.MatchString(latestVer) {
		return nil
	}
	g := &UpgradeGuide{
		Kind:  "source_build",
		Intro: fmt.Sprintf("升级要换一次镜像并重建容器，全程在你自己的电脑终端里执行几条命令，约 2 分钟；期间隧道会断几秒后自动恢复。本页面不能替你执行，只能把命令给你抄。命令里的容器名 frpc-mesh 按你自己的改；不确定就先执行 docker ps 看 NAME 列。当前版本 v%s。", latestVer),
		Note:  "配置和已下载的 frpc 版本都在宿主机目录里，重建容器不会丢；隧道会中断几秒后自动恢复。",
		Fallback: GuideStep{
			Title:    "万一升级失败或页面打不开怎么办",
			Desc:     "先看报错：下面这条会打印容器最近 100 行输出。容器起不来也不会吃掉配置——数据在宿主机目录里。要退回旧版本，重新下载旧版本镜像包 load 回来，再走一次「重建容器」那步即可。",
			Commands: []string{"docker logs --tail 100 frpc-mesh"},
		},

		asset := fmt.Sprintf(dockerAsset, latestVer)

			{

				Optional: true,
			},
			{
				Title: "下载新镜像包",
				Desc:  "约 16MB，下载后不用解压。找不到链接可以从本页「关于」卡片进仓库的 Releases 页。",

export type CopyResult = 'copied' | 'selected' | 'failed'

/**
 * 复制命令。面板常走 http://内网IP，非安全上下文下 navigator.clipboard 会被浏览器拒绝，
 * 所以三级兜底：clipboard → execCommand → 选中该行文本让用户自己按 Ctrl/Cmd+C。
 */
export async function copyText(text: string, fallbackEl?: HTMLElement | null): Promise<CopyResult> {
  try {
    if (navigator.clipboard && window.isSecureContext) {
      await navigator.clipboard.writeText(text)
      return 'copied'
    }
  } catch {
    /* 继续走兜底 */
  }
  try {
    const ta = document.createElement('textarea')
    ta.value = text
    ta.style.position = 'fixed'
    ta.style.opacity = '0'
    document.body.appendChild(ta)
    ta.select()
    const ok = document.execCommand('copy')
    document.body.removeChild(ta)
    if (ok) return 'copied'
  } catch {
    /* 继续走选中 */
  }
  if (fallbackEl && selectRange(fallbackEl)) return 'selected'
  return 'failed'
}

function selectRange(el: HTMLElement): boolean {
  const range = document.createRange()
  range.selectNodeContents(el)
  const sel = window.getSelection()
  if (!sel) return false
  sel.removeAllRanges()
  sel.addRange(range)
  return true
}
```
